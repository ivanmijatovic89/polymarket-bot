# ts-compat execution scenarios (13 §11 "compat goldens"), shared by the TS
# generator compat_gen.ts (which runs src/trading/execution/BacktestExecution.ts
# in isolation) and the Rust test exec::sim::tests::golden. The generator
# writes compat_expected.txt; the Rust simulator must render the same text.
#
# Commands (times in ms, prices and sizes as decimals):
#   scenario <name>                      fresh simulator, books and positions
#   delay <ms>                           compatLatency.delayMs (jitter 0)
#   book <up|down> bids=<p>:<s>,.. asks=<p>:<s>,..   recorded book snapshot
#   position <up|down> <qty>             delivered position (merge input)
#   place <t> <cid> <up|down> <BUY|SELL> <price> <size> <GTC|GTD|FOK> [post] [exp=<ms>] [| <next order>]...
#   cancel <t> <cid>                     cancel bound to the cid's current order
#   cancel_market <t> [up|down]          scope cancel (whole market without outcome)
#   cancel_all <t>
#   split <t> <size> / merge <t> <size>
#   tick <t>                             real in-window tick at t on the current books
